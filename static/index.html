async function startVideoCall(channelName, safeUid, role) { document.getElementById('video-call-screen').classList.add('active'); document.getElementById('chat-messages-container').innerHTML = ''; try { const res = await fetch(`/api/agora-token?channelName=${channelName}&uid=${safeUid}&role=${role}`); const data = await res.json(); if (data.status !== "success") { alert("Token error: " + data.message); document.getElementById('video-call-screen').classList.remove('active'); return; } if (agoraClient) await agoraClient.leave().catch(() => {}); agoraClient = AgoraRTC.createClient({ mode: "rtc", codec: "vp8" }); // 1. Agar host baad mein live aaye agoraClient.on("user-published", async (user, mediaType) => { await agoraClient.subscribe(user, mediaType); if (mediaType === "video") { user.videoTrack.play("remote-video"); } if (mediaType === "audio") { user.audioTrack.play(); } }); // Channel join karein await agoraClient.join(data.appId, data.channel, data.token, data.uid); // 2. Agar host pehle se hi live hai, toh turant fetch karke play karein for (let user of agoraClient.remoteUsers) { if (user.hasVideo) { await agoraClient.subscribe(user, "video"); user.videoTrack.play("remote-video"); } if (user.hasAudio) { await agoraClient.subscribe(user, "audio"); user.audioTrack.play(); } } if (role === "publisher") { document.getElementById('local-video').style.display = 'block'; localTracks.audioTrack = await AgoraRTC.createMicrophoneAudioTrack(); localTracks.videoTrack = await AgoraRTC.createCameraVideoTrack(); localTracks.videoTrack.play("local-video"); await agoraClient.publish([localTracks.audioTrack, localTracks.videoTrack]); } else { document.getElementById('local-video').style.display = 'none'; } } catch (error) { console.error("Agora join failed:", error); alert("Agora Error: " + error.message); document.getElementById('video-call-screen').classList.remove('active'); } }
